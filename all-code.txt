using System;
using System.Runtime.InteropServices;

namespace Kiero.Net;

public static unsafe class Kiero
{
    private const string Dll = "kiero_native";

    [DllImport(Dll, CallingConvention = CallingConvention.Cdecl, EntryPoint = "kiero_wrapper_init")]
    private static extern int NativeInit(int renderType);

    [DllImport(Dll, CallingConvention = CallingConvention.Cdecl, EntryPoint = "kiero_wrapper_shutdown")]
    private static extern void NativeShutdown();

    [DllImport(Dll, CallingConvention = CallingConvention.Cdecl, EntryPoint = "kiero_wrapper_get_methods_table")]
    private static extern nint NativeGetMethodsTable();

    [DllImport(Dll, CallingConvention = CallingConvention.Cdecl, EntryPoint = "kiero_wrapper_get_render_type")]
    private static extern int NativeGetRenderType();

    public static KieroStatus Init(KieroRenderType type) => (KieroStatus)NativeInit((int)type);
    public static void Shutdown() => NativeShutdown();
    public static nint GetMethodsTable() => NativeGetMethodsTable();
    public static KieroRenderType GetRenderType() => (KieroRenderType)NativeGetRenderType();

    public static nint GetMethod(int index)
    {
        nint t = GetMethodsTable();
        if (t == 0) return 0;
        return *(nint*)(t + index * IntPtr.Size);
    }
}

#nullable enable
using System;
using System.IO;
using System.Text;

namespace Kiero.Net;

public enum LogLevel
{
    Debug = 0,
    Info = 1,
    Warn = 2,
    Error = 3,
    None = 4,
}

public static class Logger
{
    private static readonly object sync = new();
    private static StreamWriter? file;
    private static LogLevel minLevel = LogLevel.Debug;
    private static string? logPath;
    private static long maxBytes = 10 * 1024 * 1024;
    private static int maxFiles = 5;

    public static LogLevel MinLevel
    {
        get => minLevel;
        set => minLevel = value;
    }

    public static void Initialize(string? directory = null, string fileName = "Kiero.log")
    {
        lock (sync)
        {
            if (file != null) return;

            try
            {
                directory ??= Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                    "Kiero.Net");

                Directory.CreateDirectory(directory);

                logPath = Path.Combine(directory, fileName);
                RotateIfNeeded(logPath, maxBytes, maxFiles);

                file = new StreamWriter(
                    new FileStream(logPath, FileMode.Append, FileAccess.Write, FileShare.Read),
                    new UTF8Encoding(false))
                {
                    AutoFlush = true,
                };

                WriteRaw(LogLevel.Info, "Logger", "========== session start ==========");
            }
            catch
            {
                file = null;
            }
        }
    }

    public static void Shutdown()
    {
        lock (sync)
        {
            if (file == null) return;

            try
            {
                WriteRaw(LogLevel.Info, "Logger", "========== session end ==========");
                file.Flush();
                file.Dispose();
            }
            catch { }
            finally
            {
                file = null;
                logPath = null;
            }
        }
    }

    public static void Debug(string tag, string message) => Write(LogLevel.Debug, tag, message);
    public static void Info(string tag, string message) => Write(LogLevel.Info, tag, message);
    public static void Warn(string tag, string message) => Write(LogLevel.Warn, tag, message);
    public static void Error(string tag, string message) => Write(LogLevel.Error, tag, message);

    public static void Error(string tag, Exception ex)
        => Write(LogLevel.Error, tag, $"{ex.GetType().Name}: {ex.Message}\n{ex.StackTrace}");

    private static void Write(LogLevel level, string tag, string message)
    {
        if (level < minLevel) return;

        lock (sync)
        {
            if (file == null) Initialize();
            WriteRaw(level, tag, message);
        }
    }

    private static void WriteRaw(LogLevel level, string tag, string message)
    {
        string stamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff");
        string prefix = level switch
        {
            LogLevel.Debug => "DBG",
            LogLevel.Info => "INF",
            LogLevel.Warn => "WRN",
            LogLevel.Error => "ERR",
            _ => "???",
        };

        string line = $"[{stamp}] {prefix} [{tag}] {message}";

        try
        {
            file?.WriteLine(line);
            file?.Flush();
        }
        catch { }

        System.Diagnostics.Debug.WriteLine(line);
    }

    private static void RotateIfNeeded(string path, long maxBytes, int maxFiles)
    {
        try
        {
            if (!File.Exists(path)) return;
            var fi = new FileInfo(path);
            if (fi.Length < maxBytes) return;

            string oldest = $"{path}.{maxFiles}";
            if (File.Exists(oldest)) File.Delete(oldest);

            for (int i = maxFiles - 1; i >= 1; i--)
            {
                string src = $"{path}.{i}";
                string dst = $"{path}.{i + 1}";
                if (File.Exists(src))
                {
                    if (File.Exists(dst)) File.Delete(dst);
                    File.Move(src, dst);
                }
            }

            File.Move(path, $"{path}.1");
        }
        catch { }
    }
}

using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;

namespace Kiero.Net;

public static unsafe class PresentHook
{
    private static nint original;
    private static long counter;

    public static bool Install(nint addr)
    {
        if (addr == 0) return false;
        if (MinHook.MH_Initialize() != 0) return false;

        void* detour = (void*)(delegate* unmanaged[Stdcall]<nint, uint, uint, int>)&Detour;
        void* orig = null;
        int hr = MinHook.MH_CreateHook((void*)addr, detour, &orig);
        if (hr != 0) return false;

        original = (nint)orig;
        MinHook.MH_EnableHook((void*)addr);
        return true;
    }

    [UnmanagedCallersOnly(CallConvs = new[] { typeof(CallConvStdcall) })]
    private static int Detour(nint sc, uint sync, uint flags)
    {
        long n = ++counter;
        if (n == 1 || n % 300 == 0)
            Logger.Info("Present", $"#{n} sc=0x{sc:X}");

        return ((delegate* unmanaged[Stdcall]<nint, uint, uint, int>)original)(sc, sync, flags);
    }
}

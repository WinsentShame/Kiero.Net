internal sealed unsafe class Dx11Backend
{

    private const int VT_GetDevice = 3;
    private const int VT_GetImmediateContext = 40;
    private const int VT_CreateRenderTargetView = 9;
    private const int VT_OMSetRenderTargets = 33;

    private nint swapChain;
    private nint device;
    private nint context;
    private nint rtv;
    private nint rtvBb;
    private bool ready;
    private long frameCounter;

    public bool isReady => ready;

    public bool Init(nint sc)
    {
        swapChain = sc;
        Logger.Info("Dx11", $"Init sc=0x{sc:X}");

        nint bb = GetBackBuffer(sc, 0);
        if (bb == 0) { Logger.Error("Dx11", "backbuffer=0"); return false; }

        nint dev = 0;
        var getDevice = (delegate* unmanaged[Stdcall]<nint, nint*, int>)((*(void***)bb)[VT_GetDevice]);
        int hr = getDevice(bb, &dev);
        Com.Release(bb);
        if (hr < 0 || dev == 0) { Logger.Error("Dx11", $"GetDevice hr=0x{hr:X}"); return false; }
        device = dev;
        Logger.Info("Dx11", $"device=0x{dev:X}");

        nint ctx = 0;

        getCtx(dev, &ctx);
        if (ctx == 0) { Logger.Error("Dx11", "no context"); return false; }
        context = ctx;
        Logger.Info("Dx11", $"context=0x{ctx:X}");

        if (ImGui.ImplDX11Init(device, context) == 0)
        {
            Logger.Error("Dx11", "ImplDX11Init failed");
            return false;
        }

        ready = true;
        return true;
    }

    public void NewFrame() => ImGui.ImplDX11NewFrame();

    public void ReleaseRenderTarget() { if (rtv != 0) { Com.Release(rtv); rtv = 0; } rtvBb = 0; }

    public void RenderDrawData(nint drawData)
    {
        if (!ready || drawData == 0) return;

        uint IDX = GetCurrentBackBufferIndex(swapChain);
        nint bb = GetBackBuffer(swapChain, IDX);
        if (bb == 0) return;

        try
        {
            if (rtv == 0 || rtvBb != bb)
            {
                if (rtv != 0) { Com.Release(rtv); rtv = 0; }

                nint newRtv = 0;
                var createRtv = (delegate* unmanaged[Stdcall]<nint, nint, nint, nint*, int>)
                    ((*(void***)device)[VT_CreateRenderTargetView]);
                int hr = createRtv(device, bb, 0, &newRtv);
                if (hr < 0 || newRtv == 0) return;
                rtv = newRtv;
                rtvBb = bb;
            }

            nint rtvHandle = rtv;
            var omSet = (delegate* unmanaged[Stdcall]<nint, uint, nint*, nint, void>)
                ((*(void***)context)[VT_OMSetRenderTargets]);
            omSet(context, 1, &rtvHandle, 0);

            ImGui.ImplDX11RenderDrawData(drawData);

            long N = ++frameCounter;
            if (N == 1 || N % 600 == 0)
                Logger.Info("Dx11", $"frame={N} idx={IDX} bb=0x{bb:X} rtv=0x{rtv:X}");
        }
        finally
        {
            Com.Release(bb);
        }
    }

    public void Shutdown()
    {
        ReleaseRenderTarget();
        if (ready)
        {
            try { ImGui.ImplDX11Shutdown(); } catch { }
            ready = false;
        }
        if (context != 0) { Com.Release(context); context = 0; }
        if (device != 0) { Com.Release(device); device = 0; }
    }

using System;
using System.Diagnostics;

namespace Kiero.Net;

public static unsafe class Overlay
{
    private const uint dxgiPresentTest = 0x1;

    public static Action onDraw { get; set; }
    public static Action onInput { get; set; }
    public static bool isReady => ready;
    internal static nint Dx11SwapChain => swapChain;
    internal static Dx11Backend Dx11Backend => backend;

    private static readonly object gate = new();
    private static nint context;
    private static nint swapChain;
    private static volatile bool ready;
    private static volatile bool initAttempted;
    private static volatile bool started;
    private static long lastTicks;
    private static bool useWin32Backend;
    private static int skipLogCount;

    private static OverlayMode mode = OverlayMode.Auto;
    private static Dx11Backend backend;

    public static void Initialize(OverlayMode overlayMode = OverlayMode.Auto)
    {
        if (started) return;
        started = true;
        mode = overlayMode;
        Logger.Info("Overlay", $"Initialize mode={overlayMode}");
    }

    internal static void OnPresentInternal(nint sc, uint flags)
    {
        if ((flags & dxgiPresentTest) != 0) return;
        if (sc == 0) return;

        lock (gate)
        {
            if (!ready && !InitImGui(sc)) return;
            if (sc != swapChain) return;
            RenderFrame(sc);
        }
    }

    private static bool InitImGui(nint sc)
    {
        if (initAttempted) return false;

        byte* desc = stackalloc byte[128];
        var getDesc = (delegate* unmanaged[Stdcall]<nint, void*, int>)((*(void***)sc)[12]);
        if (getDesc(sc, desc) < 0) return false;

        uint w = *(uint*)(desc + 0);
        uint h = *(uint*)(desc + 4);
        nint hwnd = *(nint*)(desc + 48);

        if (w < 64 || h < 64) return false;

        bool wantWin32;
        switch (mode)
        {
            case OverlayMode.Win32:
                wantWin32 = true;
                break;
            case OverlayMode.Uwp:
                wantWin32 = false;
                break;
            default:
                wantWin32 = hwnd != 0;
                break;
        }

        if (mode == OverlayMode.Win32 && hwnd == 0)
        {
            if (skipLogCount++ < 3)
                Logger.Warn("Overlay", $"mode=Win32 but hwnd=0 sc=0x{sc:X}");
            return false;
        }

        initAttempted = true;
        Logger.Info("Overlay", $"InitImGui sc=0x{sc:X} {w}x{h} hwnd=0x{hwnd:X} mode={mode}");

        try
        {
            context = ImGui.CreateContext(0);
            if (context == 0) { Logger.Error("Overlay", "no context"); initAttempted = false; return false; }

            ImGui.SetCurrentContext(context);
            ImGui.StyleColorsDark(0);

            nint io = ImGui.GetIO();
            if (io == 0) { Logger.Error("Overlay", "no io"); initAttempted = false; return false; }
            ImGui.IOSetIniFilename(io, null);

            if (wantWin32)
            {
                if (ImGui.ImplWin32Init(hwnd) != 0)
                {
                    useWin32Backend = true;
                    Logger.Info("Overlay", "Win32 backend initialized");
                }
                else if (mode == OverlayMode.Auto)
                {
                    Logger.Warn("Overlay", "Win32Init failed, falling back to UWP mode");
                    useWin32Backend = false;
                }
                else
                {
                    Logger.Error("Overlay", "Win32Init failed");
                    initAttempted = false;
                    return false;
                }
            }
            else
            {
                useWin32Backend = false;
                Logger.Info("Overlay", "UWP mode: skipping Win32 backend");
            }

            backend = new Dx11Backend();
            if (!backend.Init(sc))
            {
                Logger.Error("Overlay", "Dx11Backend.Init failed");
                initAttempted = false;
                return false;
            }

            swapChain = sc;
            lastTicks = Stopwatch.GetTimestamp();
            ready = true;
            Logger.Info("Overlay", $"ImGui ready (win32={useWin32Backend})");
            return true;
        }
        catch (Exception ex)
        {
            Logger.Error("Overlay", ex);
            initAttempted = false;
            return false;
        }
    }



    private static void RenderFrame(nint sc)
    {
        float w, h;
        if (!backend.TryGetSize(sc, out w, out h)) return;
        if (w <= 0 || h <= 0) return;

        long now = Stopwatch.GetTimestamp();
        float dt = (float)((now - lastTicks) / (double)Stopwatch.Frequency);
        lastTicks = now;
        if (dt <= 0f || dt > 0.5f) dt = 1f / 60f;

        ImGui.SetCurrentContext(context);
        nint io = ImGui.GetIO();

        if (useWin32Backend)
            ImGui.ImplWin32NewFrame();

        backend.NewFrame();

        ImGui.IOSetDisplaySize(io, w, h);
        ImGui.IOSetDeltaTime(io, dt);

        try { onInput?.Invoke(); } catch { }

        ImGui.NewFrame();
        try { onDraw?.Invoke(); } catch { }
        ImGui.Render();

        nint dd = ImGui.GetDrawData();
        if (dd == 0) return;
        if (ImGui.DrawDataVtxCount(dd) <= 0) return;

        backend.RenderDrawData(dd);
    }
}

#nullable enable
using System;
using System.IO;
using System.Text;

namespace Kiero.Net;

public enum LogLevel
{
    Debug = 0,
    Info = 1,
    Warn = 2,
    Error = 3,
    None = 4,
}

public static class Logger
{
    private static readonly object sync = new();
    private static StreamWriter? file;
    private static string? logPath;
    private static long maxBytes = 10 * 1024 * 1024;
    private static int maxFiles = 5;

    public static LogLevel minLevel { get; set; } = LogLevel.Debug;

    public static void Initialize(string? directory = null, string fileName = "Kiero.log")
    {
        lock (sync)
        {
            if (file != null) return;

            try
            {
                directory ??= Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                    "Kiero.Net");

                Directory.CreateDirectory(directory);

                logPath = Path.Combine(directory, fileName);
                RotateIfNeeded(logPath, maxBytes, maxFiles);

                file = new StreamWriter(
                    new FileStream(logPath, FileMode.Append, FileAccess.Write, FileShare.Read),
                    new UTF8Encoding(false))
                {
                    AutoFlush = true,
                };

                WriteRaw(LogLevel.Info, "Logger", "========== session start ==========");
            }
            catch
            {
                file = null;
            }
        }
    }

    public static void Shutdown()
    {
        lock (sync)
        {
            if (file == null) return;

            try
            {
                WriteRaw(LogLevel.Info, "Logger", "========== session end ==========");
                file.Flush();
                file.Dispose();
            }
            catch { }
            finally
            {
                file = null;
                logPath = null;
            }
        }
    }

    public static void Debug(string tag, string message) => Write(LogLevel.Debug, tag, message);
    public static void Info(string tag, string message) => Write(LogLevel.Info, tag, message);
    public static void Warn(string tag, string message) => Write(LogLevel.Warn, tag, message);
    public static void Error(string tag, string message) => Write(LogLevel.Error, tag, message);

    public static void Error(string tag, Exception ex)
        => Write(LogLevel.Error, tag, $"{ex.GetType().Name}: {ex.Message}\n{ex.StackTrace}");

    private static void Write(LogLevel level, string tag, string message)
    {
        if (level < minLevel) return;

        lock (sync)
        {
            if (file == null) Initialize();
            WriteRaw(level, tag, message);
        }
    }

    private static void WriteRaw(LogLevel level, string tag, string message)
    {
        string stamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff");
        string prefix = level switch
        {
            LogLevel.Debug => "DBG",
            LogLevel.Info => "INF",
            LogLevel.Warn => "WRN",
            LogLevel.Error => "ERR",
            _ => "???",
        };

        string line = $"[{stamp}] {prefix} [{tag}] {message}";

        try
        {
            file?.WriteLine(line);
            file?.Flush();
        }
        catch { }

        System.Diagnostics.Debug.WriteLine(line);
    }

    private static void RotateIfNeeded(string path, long maxBytes, long maxFiles)
    {
        try
        {
            if (!File.Exists(path)) return;
            var fi = new FileInfo(path);
            if (fi.Length < maxBytes) return;

            string oldest = $"{path}.{maxFiles}";
            if (File.Exists(oldest)) File.Delete(oldest);

            for (int i = (int)maxFiles - 1; i >= 1; i--)
            {
                string src = $"{path}.{i}";
                string dst = $"{path}.{i + 1}";
                if (File.Exists(src))
                {
                    if (File.Exists(dst)) File.Delete(dst);
                    File.Move(src, dst);
                }
            }

            File.Move(path, $"{path}.1");
        }
        catch { }
    }
}

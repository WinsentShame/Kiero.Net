using System;

namespace Kiero.Net;

internal sealed unsafe class Dx11Backend
{
    private const int VT_GetDevice = 3;
    private const int VT_CreateRenderTargetView = 9;
    private const int VT_OMSetRenderTargets = 33;
    private const int VT_GetImmediateContext = 40;

    private nint swapChain;
    private nint device;
    private nint context;
    private nint rtv;
    private nint rtvBb;
    private bool ready;
    private long frameCounter;

    public bool isReady => ready;

    public bool Init(nint sc)
    {
        swapChain = sc;
        Logger.Info("Dx11", $"Init sc=0x{sc:X}");

        nint bb = Dxgi.GetBackBuffer(sc, 0, Guids.IID_ID3D11Texture2D);
        if (bb == 0) { Logger.Error("Dx11", "backbuffer=0"); return false; }

        nint dev = 0;
        var getDevice = (delegate* unmanaged[Stdcall]<nint, nint*, int>)((*(void***)bb)[VT_GetDevice]);
        int hr = getDevice(bb, &dev);
        Com.Release(bb);
        if (hr < 0 || dev == 0) { Logger.Error("Dx11", $"GetDevice hr=0x{hr:X}"); return false; }
        device = dev;
        Logger.Info("Dx11", $"device=0x{dev:X}");

        nint ctx = 0;
        var getCtx = (delegate* unmanaged[Stdcall]<nint, nint*, void>)((*(void***)dev)[VT_GetImmediateContext]);
        getCtx(dev, &ctx);
        if (ctx == 0) { Logger.Error("Dx11", "no context"); return false; }
        context = ctx;
        Logger.Info("Dx11", $"context=0x{ctx:X}");

        if (ImGui.ImplDX11Init(device, context) == 0)
        {
            Logger.Error("Dx11", "ImplDX11Init failed");
            return false;
        }

        ready = true;
        return true;
    }

    public void NewFrame() => ImGui.ImplDX11NewFrame();

    public void ReleaseRenderTarget() { if (rtv != 0) { Com.Release(rtv); rtv = 0; } rtvBb = 0; }

    public void RenderDrawData(nint drawData)
    {
        if (!ready || drawData == 0) return;

        uint IDX = Dxgi.GetCurrentBackBufferIndex(swapChain);
        nint bb = Dxgi.GetBackBuffer(swapChain, IDX, Guids.IID_ID3D11Texture2D);
        if (bb == 0) return;

        try
        {
            if (rtv == 0 || rtvBb != bb)
            {
                if (rtv != 0) { Com.Release(rtv); rtv = 0; }

                nint newRtv = 0;
                var createRtv = (delegate* unmanaged[Stdcall]<nint, nint, nint, nint*, int>)
                    ((*(void***)device)[VT_CreateRenderTargetView]);
                int hr = createRtv(device, bb, 0, &newRtv);
                if (hr < 0 || newRtv == 0) return;
                rtv = newRtv;
                rtvBb = bb;
            }

            nint rtvHandle = rtv;
            var omSet = (delegate* unmanaged[Stdcall]<nint, uint, nint*, nint, void>)
                ((*(void***)context)[VT_OMSetRenderTargets]);
            omSet(context, 1, &rtvHandle, 0);

            ImGui.ImplDX11RenderDrawData(drawData);

            long N = ++frameCounter;
            if (N == 1 || N % 600 == 0)
                Logger.Info("Dx11", $"frame={N} idx={IDX} bb=0x{bb:X} rtv=0x{rtv:X}");
        }
        finally
        {
            Com.Release(bb);
        }
    }

    public void Shutdown()
    {
        ReleaseRenderTarget();
        if (ready)
        {
            try { ImGui.ImplDX11Shutdown(); } catch { }
            ready = false;
        }
        if (context != 0) { Com.Release(context); context = 0; }
        if (device != 0) { Com.Release(device); device = 0; }
    }
}

            frameIndex = (fi + 1) % FRAMES_IN_FLIGHT;

            long N = ++frameCounter;
            if (N == 1 || N % 600 == 0)
                Logger.Info("Dx12", $"frame={N} idx={idx} bb=0x{backbuffer:X}");
        }
        finally
        {
            Com.Release(backbuffer);
        }
    }

public static unsafe class Dx12ExecuteCommandListsFinder
{
    public static bool TryGet(out nint executeCommandLists)
    {
        executeCommandLists = 0;

        Guid iidFactory = Guids.IID_IDXGIFactory2;
        nint factory = 0;
        int hr = D3D12.CreateDXGIFactory2(0, &iidFactory, &factory);
        if (hr < 0 || factory == 0) { Logger.Error("Dx12Finder", $"CreateDXGIFactory2 hr=0x{hr:X8}"); return false; }

        nint adapter = 0;
        nint device = 0;
        nint queue = 0;

        try
        {
            nint vtFactory = *(nint*)factory;
            var enumAdapters = (delegate* unmanaged[Stdcall]<nint, uint, nint*, int>)(*(nint*)(vtFactory + 12 * IntPtr.Size));

            byte* desc = stackalloc byte[0x140];
            for (uint i = 0; i < 8; i++)
            {
                nint a = 0;
                if (enumAdapters(factory, i, &a) < 0 || a == 0) break;

                nint vtA = *(nint*)a;
                var getDesc = (delegate* unmanaged[Stdcall]<nint, void*, int>)(*(nint*)(vtA + 10 * IntPtr.Size));

                for (int b = 0; b < 0x140; b++) desc[b] = 0;

                if (getDesc(a, desc) >= 0)
                {
                    uint flags = *(uint*)(desc + 0x130);
                    if ((flags & 0x2) == 0) { adapter = a; break; }
                }
                Com.Release(a);
            }

            if (adapter == 0) { Logger.Error("Dx12Finder", "no hardware adapter found"); return false; }

            Guid iidDev = Guids.IID_ID3D12Device;
            hr = D3D12.D3D12CreateDevice(adapter, D3D12.FEATURE_LEVEL_11_0, &iidDev, &device);
            if (hr < 0 || device == 0) { Logger.Error("Dx12Finder", $"D3D12CreateDevice hr=0x{hr:X8}"); return false; }

            var qdesc = new D3D12Structs.D3D12_COMMAND_QUEUE_DESC { Type = D3D12Structs.COMMAND_LIST_TYPE_DIRECT };
            nint vtDev = *(nint*)device;
            var createQueue = (delegate* unmanaged[Stdcall]<nint, D3D12Structs.D3D12_COMMAND_QUEUE_DESC*, Guid*, nint*, int>)(*(nint*)(vtDev + 8 * IntPtr.Size));
            Guid iidQueue = Guids.IID_ID3D12CommandQueue;
            hr = createQueue(device, &qdesc, &iidQueue, &queue);
            if (hr < 0 || queue == 0) { Logger.Error("Dx12Finder", $"CreateCommandQueue hr=0x{hr:X8}"); return false; }

            nint vtQueue = *(nint*)queue;
            nint execCmdListsAddr = *(nint*)(vtQueue + 10 * IntPtr.Size);
            if (execCmdListsAddr != 0 && !ExecuteCommandListsHook.IsInstalled)
            {
                if (ExecuteCommandListsHook.Install(execCmdListsAddr))
                {
                    Logger.Info("Dx12", "ExecuteCommandLists hook installed (from game queue)");
                    executeCommandLists = execCmdListsAddr; // <--- ВОТ ЭТА СТРОКА!
                }
                else
                {
                    Logger.Warn("Dx12", "ExecuteCommandLists hook failed");
                }
            }

            return executeCommandLists != 0;
        }
        finally
        {
            if (queue != 0) Com.Release(queue);
            if (device != 0) Com.Release(device);
            if (adapter != 0) Com.Release(adapter);
            if (factory != 0) Com.Release(factory);
        }
    }
}

using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using System.Threading;

namespace Kiero.Net;

public static unsafe class ExecuteCommandListsHook
{
    private const int COMMAND_LIST_TYPE_DIRECT = 0;

    private static nint original;
    private static nint lastQueue;
    private static int installed;

    public static nint LastQueue => lastQueue;
    public static bool IsInstalled => installed != 0;

    public static bool Install(nint addr)
    {
        if (addr == 0) return false;
        if (Interlocked.CompareExchange(ref installed, 1, 0) != 0) return true;

        void* detour = (delegate* unmanaged[Stdcall]<nint, uint, nint*, void>)&Detour;
        void* orig = null;
        if (MinHook.CreateHook((void*)addr, detour, &orig) != 0) { installed = 0; return false; }
        original = (nint)orig;
        if (MinHook.EnableHook((void*)addr) != 0) { installed = 0; return false; }
        return true;
    }

    [UnmanagedCallersOnly(CallConvs = new[] { typeof(CallConvStdcall) })]
    private static void Detour(nint queue, uint count, nint* lists)
    {
        if (queue != 0 && lastQueue == 0)
        {
            int type = D3D12Shim.QueueGetType(queue);
            if (type == COMMAND_LIST_TYPE_DIRECT)
            {
                lastQueue = queue;
                Logger.Info("ECLHook", $"captured direct queue=0x{queue:X}");
            }
        }

        ((delegate* unmanaged[Stdcall]<nint, uint, nint*, void>)original)(queue, count, lists);
    }
}

public static unsafe class PresentHook

    public static long presentCallCount => Interlocked.Read(ref presentCalls);
    public static long present1CallCount => Interlocked.Read(ref present1Calls);

    public static bool Install(nint addr)
    {
        if (addr == 0) return false;
        return InstallOne(addr, false);
    }

    public static bool InstallPresent1(nint addr)
    {
        if (addr == 0) return false;
        return InstallOne(addr, true);
    }

    private static bool InstallOne(nint addr, bool isPresent1)
    {
        void* detour = isPresent1
            ? (delegate* unmanaged[Stdcall]<nint, uint, uint, void*, int>)&DetourPresent1
            : (delegate* unmanaged[Stdcall]<nint, uint, uint, int>)&DetourPresent;

        void* orig = null;
        int hr = MinHook.CreateHook((void*)addr, detour, &orig);
        if (hr != 0)
        {
            Logger.Warn("Present", $"CreateHook({(isPresent1 ? "Present1" : "Present")}) hr={hr}");
            return false;
        }

        if (isPresent1) originalPresent1 = (nint)orig;
        else originalPresent = (nint)orig;

        int hr2 = MinHook.EnableHook((void*)addr);
        if (hr2 != 0)
        {
            Logger.Warn("Present", $"EnableHook hr={hr2}");
            return false;
        }
        return true;
    }

    [UnmanagedCallersOnly(CallConvs = new[] { typeof(CallConvStdcall) })]
    private static int DetourPresent(nint sc, uint sync, uint flags)
    {
        long n = Interlocked.Increment(ref presentCalls);
        if (n <= 3 || n % 300 == 0)
            Logger.Info("Present", $"Present #{n} sc=0x{sc:X} flags=0x{flags:X}");

        try { Overlay.OnPresentInternal(sc, flags); }
        catch (System.Exception ex) { Logger.Error("Present", ex); }

        return ((delegate* unmanaged[Stdcall]<nint, uint, uint, int>)originalPresent)(sc, sync, flags);
    }

    [UnmanagedCallersOnly(CallConvs = new[] { typeof(CallConvStdcall) })]
    private static int DetourPresent1(nint sc, uint sync, uint flags, void* presentParams)
    {
        long n = Interlocked.Increment(ref present1Calls);
        if (n <= 3 || n % 300 == 0)
            Logger.Info("Present", $"Present1 #{n} sc=0x{sc:X} flags=0x{flags:X}");

        try { Overlay.OnPresentInternal(sc, flags); }
        catch (System.Exception ex) { Logger.Error("Present", ex); }

        return ((delegate* unmanaged[Stdcall]<nint, uint, uint, void*, int>)originalPresent1)(sc, sync, flags, presentParams);
    }
}

using System;
using System.Threading;

namespace Kiero.Net;

public static class Bootstrap
{
    public static bool Start(TimeSpan? moduleTimeout = null)
    {
        int waited = 0;
        int limit = (int)(moduleTimeout ?? TimeSpan.FromSeconds(60)).TotalMilliseconds;

        bool has11 = false, has12 = false;
        while (waited < limit)
        {
            has11 = Win32.GetModuleHandleW("d3d11.dll") != 0;
            has12 = Win32.GetModuleHandleW("d3d12.dll") != 0;
            if (has11 || has12) break;
            Thread.Sleep(200);
            waited += 200;
        }

        Logger.Info("Bootstrap", $"after wait {waited}ms: d3d11={has11} d3d12={has12}");
        if (!has11 && !has12) { Logger.Error("Bootstrap", "no d3d11/d3d12"); return false; }

        var type = has12 ? KieroRenderType.D3D12 : KieroRenderType.D3D11;
        var st = Kiero.Init(type);
        if (st != KieroStatus.Success) { Logger.Error("Bootstrap", $"Kiero.Init({type}) -> {st}"); return false; }

        if (MinHook.Initialize() != 0) { Logger.Error("Bootstrap", "MinHook.Initialize failed"); return false; }
        Logger.Info("Bootstrap", "MinHook initialized");

        var api = Kiero.GetRenderType();
        Logger.Info("Bootstrap", $"Kiero api={api}");
        return api == KieroRenderType.D3D12 ? InstallDx12() : InstallDx11();
    }

    private static bool InstallDx11()
    {
        if (!Dx11PresentFinder.TryGet(out nint presentAddr, out nint present1Addr, out nint resizeAddr))
        {
            Logger.Error("Bootstrap", "Dx11PresentFinder failed");
            return false;
        }

        Logger.Info("Bootstrap", $"Dx11 Present=0x{presentAddr:X} Present1=0x{present1Addr:X} ResizeBuffers=0x{resizeAddr:X}");

        if (resizeAddr != 0 && ResizeBuffersHook.Install(resizeAddr))
            Logger.Info("Bootstrap", "ResizeBuffers hook installed");
        else
            Logger.Warn("Bootstrap", "ResizeBuffers hook not installed");

        if (presentAddr == 0)
        {
            Logger.Error("Bootstrap", "Present address is null");
            return false;
        }

        if (!PresentHook.Install(presentAddr))
        {
            Logger.Error("Bootstrap", "Present hook install failed");
            return false;
        }
        Logger.Info("Bootstrap", "Present hook installed");

        if (present1Addr != 0 && present1Addr != presentAddr)
        {
            if (PresentHook.InstallPresent1(present1Addr))
                Logger.Info("Bootstrap", "Present1 hook installed");
            else
                Logger.Warn("Bootstrap", "Present1 hook not installed");
        }

        return true;
    }

    private static bool InstallDx12()
    {
        if (!Dx11PresentFinder.TryGet(out nint presentAddr, out nint present1Addr, out nint resizeAddr))
        {
            Logger.Error("Bootstrap", "Dx11PresentFinder failed");
            return false;
        }

        Logger.Info("Bootstrap", $"Dx12 via Dx11 vtable Present=0x{presentAddr:X} Present1=0x{present1Addr:X} ResizeBuffers=0x{resizeAddr:X}");
        if (Dx12ExecuteCommandListsFinder.TryGet(out nint execCmdListsAddr))
        {
            Logger.Info("Bootstrap", "ExecuteCommandLists hook installed");
        }
        else
        {
            Logger.Warn("Bootstrap", "Dx12ExecuteCommandListsFinder failed (Hook may still be active)");
        }

        if (resizeAddr != 0 && ResizeBuffersHook.Install(resizeAddr))
            Logger.Info("Bootstrap", "ResizeBuffers hook installed");
        else
            Logger.Warn("Bootstrap", "ResizeBuffers hook not installed");

        if (presentAddr == 0)
        {
            Logger.Error("Bootstrap", "Present address is null");
            return false;
        }

        if (!PresentHook.Install(presentAddr))
        {
            Logger.Error("Bootstrap", "Present hook install failed");
            return false;
        }
        Logger.Info("Bootstrap", "Present hook installed");

        if (present1Addr != 0 && present1Addr != presentAddr)
        {
            if (PresentHook.InstallPresent1(present1Addr))
                Logger.Info("Bootstrap", "Present1 hook installed");
            else
                Logger.Warn("Bootstrap", "Present1 hook not installed");
        }

        return true;
    }
}

using System;
using System.Diagnostics;

namespace Kiero.Net;

public static unsafe class Overlay
{
    private const uint dxgiPresentTest = 0x1;

    public static Action onDraw { get; set; }
    public static Action onInput { get; set; }

    internal static nint Dx11SwapChain => swapChain;
    internal static Dx11Backend Dx11Backend => backend11;

    public static bool isReady => ready;

    private static readonly object gate = new();
    private static nint context;
    private static nint swapChain;
    private static volatile bool ready;
    private static volatile bool initAttempted;
    private static volatile bool started;
    private static long lastTicks;
    private static bool queueWaitLogged;

    private static RenderApi api = RenderApi.None;
    private static Dx11Backend backend11;
    private static Dx12Backend backend12;

    public static void Initialize()
    {
        if (started) return;
        started = true;
        Logger.Info("Overlay", "Initialize");
    }

    internal static void OnPresentInternal(nint sc, uint flags)
    {
        if ((flags & dxgiPresentTest) != 0) return;
        if (sc == 0) return;

        lock (gate)
        {
            if (!ready && !InitImGui(sc)) return;
            if (sc != swapChain) return;
            RenderFrame(sc);
        }
    }

    private static bool InitImGui(nint sc)
    {
        if (initAttempted) return false;

        if (Kiero.GetRenderType() == KieroRenderType.D3D12 && ExecuteCommandListsHook.LastQueue == 0)
        {
            if (!queueWaitLogged)
            {
                queueWaitLogged = true;
                Logger.Info("Overlay", "DX12 queue not captured yet, defer");
            }
            return false;
        }

        initAttempted = true;

        if (!Dxgi.TryGetDesc(sc, out uint w, out uint h, out uint fmt, out nint hwnd))
        {
            Logger.Error("Overlay", "GetDesc failed");
            return false;
        }

        if (w < 64 || h < 64) { Logger.Error("Overlay", $"too small {w}x{h}"); return false; }

        Logger.Info("Overlay", $"InitImGui sc=0x{sc:X} {w}x{h} hwnd=0x{hwnd:X}");

        context = ImGui.CreateContext(0);
        if (context == 0) { Logger.Error("Overlay", "no context"); return false; }

        ImGui.SetCurrentContext(context);
        ImGui.StyleColorsDark(0);

        nint io = ImGui.GetIO();
        if (io == 0) { Logger.Error("Overlay", "no io"); return false; }
        ImGui.IOSetIniFilename(io, null);

        var kieroType = Kiero.GetRenderType();
        if (kieroType == KieroRenderType.D3D12)
        {
            backend12 = new Dx12Backend();
            if (!backend12.Init(sc, (int)fmt))
            {
                Logger.Error("Overlay", "Dx12Backend.Init failed");
                backend12 = null;
                initAttempted = false;
                return false;
            }
            api = RenderApi.Dx12;
        }
        else
        {
            backend11 = new Dx11Backend();
            if (!backend11.Init(sc))
            {
                Logger.Error("Overlay", "Dx11Backend.Init failed");
                backend11 = null;
                return false;
            }
            api = RenderApi.Dx11;
        }

        swapChain = sc;
        lastTicks = Stopwatch.GetTimestamp();
        ready = true;
        Logger.Info("Overlay", $"ImGui ready (api={api})");
        return true;
    }

    private static void RenderFrame(nint sc)
    {
        if (!Dxgi.TryGetSize(sc, out uint wu, out uint hu)) return;
        if (wu == 0 || hu == 0) return;
        float w = wu, h = hu;

        long now = Stopwatch.GetTimestamp();
        float dt = (float)((now - lastTicks) / (double)Stopwatch.Frequency);
        lastTicks = now;
        if (dt <= 0f || dt > 0.5f) dt = 1f / 60f;

        ImGui.SetCurrentContext(context);
        nint io = ImGui.GetIO();

        if (api == RenderApi.Dx11) backend11.NewFrame();
        else backend12.NewFrame();

        ImGui.IOSetDisplaySize(io, w, h);
        ImGui.IOSetDeltaTime(io, dt);

        try { onInput?.Invoke(); } catch { }

        ImGui.NewFrame();
        try { onDraw?.Invoke(); } catch { }
        ImGui.Render();

        nint dd = ImGui.GetDrawData();
        if (dd == 0) return;
        if (ImGui.DrawDataVtxCount(dd) <= 0) return;

        if (api == RenderApi.Dx11) backend11.RenderDrawData(dd);
        else backend12.RenderDrawData(dd);
    }
}

internal enum RenderApi
{
    None = 0,
    Dx11 = 1,
    Dx12 = 2,
}

    public static bool Combo(string label, ref int current, string[] items, int popupMaxHeight = -1)
    {
        int total = 1;
        for (int i = 0; i < items.Length; i++) total += System.Text.Encoding.UTF8.GetByteCount(items[i]) + 1;

        Span<byte> buf = total <= 2048 ? stackalloc byte[total] : new byte[total];
        int pos = 0;
        for (int i = 0; i < items.Length; i++)
        {
            pos += System.Text.Encoding.UTF8.GetBytes(items[i], buf.Slice(pos));
            buf[pos++] = 0;
        }
        buf[pos] = 0;

        fixed (byte* pb = buf)
        fixed (int* pc = &current)
            return ImGui.ComboZeros(Utf8.Of(label), pc, pb, popupMaxHeight) != 0;

#nullable enable
using System;
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using System.Text;

namespace Kiero.Net;

public static unsafe class Utf8
{
    private const int ARENA_SIZE = 32 * 1024;
    private const int BIG_SLOTS = 4;

    [ThreadStatic] private static byte[]? ARENA;
    [ThreadStatic] private static int OFFSET;
    [ThreadStatic] private static byte[][]? BIG;
    [ThreadStatic] private static int BIG_INDEX;

    public static byte* Of(string? text)
    {
        if (text == null) return null;

        int need = Encoding.UTF8.GetByteCount(text) + 1;

        if (need <= ARENA_SIZE)
        {
            ARENA ??= GC.AllocateUninitializedArray<byte>(ARENA_SIZE, pinned: true);
            if (OFFSET + need > ARENA_SIZE) OFFSET = 0;

            int start = OFFSET;
            int n = Encoding.UTF8.GetBytes(text, 0, text.Length, ARENA, start);
            ARENA[start + n] = 0;
            OFFSET = start + need;
            return (byte*)Unsafe.AsPointer(ref ARENA[start]);
        }

        BIG ??= new byte[BIG_SLOTS][];
        byte[] big = GC.AllocateUninitializedArray<byte>(need, pinned: true);
        int m = Encoding.UTF8.GetBytes(text, 0, text.Length, big, 0);
        big[m] = 0;
        BIG[BIG_INDEX] = big;
        BIG_INDEX = (BIG_INDEX + 1) % BIG_SLOTS;
        return (byte*)Unsafe.AsPointer(ref big[0]);
    }

    public static string? FromPtr(byte* p) => p == null ? null : Marshal.PtrToStringUTF8((nint)p);

    public static void WriteZ(string? s, Span<byte> dst)
    {
        dst.Clear();
        if (string.IsNullOrEmpty(s)) return;

        int max = dst.Length - 1;
        int chars = s.Length;
        while (chars > 0 && Encoding.UTF8.GetByteCount(s.AsSpan(0, chars)) > max) chars--;
        Encoding.UTF8.GetBytes(s.AsSpan(0, chars), dst);
    }

    public static string ReadZ(ReadOnlySpan<byte> buf)
    {
        int len = buf.IndexOf((byte)0);
        if (len < 0) len = buf.Length;
        return Encoding.UTF8.GetString(buf.Slice(0, len));
    }

    public static byte B(bool v) => (byte)(v ? 1 : 0);

    public static void Need(int have, int need, string what)
    {
        if (have < need) throw new ArgumentException($"{what}: нужно минимум {need} элемента(ов), передано {have}");
    }
}

#nullable enable
using System;
using System.IO;
using System.Text;

namespace Kiero.Net;

public enum LogLevel
{
    Debug = 0,
    Info = 1,
    Warn = 2,
    Error = 3,
    None = 4,
}

public static class Logger
{
    private static readonly object sync = new();
    private static StreamWriter? file;
    private static string? logPath;
    private static long maxBytes = 10 * 1024 * 1024;
    private static int maxFiles = 5;

    public static LogLevel minLevel { get; set; } = LogLevel.Debug;

    public static void Initialize(string? directory = null, string fileName = "Kiero.log")
    {
        lock (sync)
        {
            if (file != null) return;

            try
            {
                directory ??= Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                    "Kiero.Net");

                Directory.CreateDirectory(directory);

                logPath = Path.Combine(directory, fileName);
                RotateIfNeeded(logPath, maxBytes, maxFiles);

                file = new StreamWriter(
                    new FileStream(logPath, FileMode.Append, FileAccess.Write, FileShare.Read),
                    new UTF8Encoding(false))
                {
                    AutoFlush = true,
                };

                WriteRaw(LogLevel.Info, "Logger", "========== session start ==========");
            }
            catch
            {
                file = null;
            }
        }
    }

    public static void Shutdown()
    {
        lock (sync)
        {
            if (file == null) return;

            try
            {
                WriteRaw(LogLevel.Info, "Logger", "========== session end ==========");
                file.Flush();
                file.Dispose();
            }
            catch { }
            finally
            {
                file = null;
                logPath = null;
            }
        }
    }

    public static void Debug(string tag, string message) => Write(LogLevel.Debug, tag, message);
    public static void Info(string tag, string message) => Write(LogLevel.Info, tag, message);
    public static void Warn(string tag, string message) => Write(LogLevel.Warn, tag, message);
    public static void Error(string tag, string message) => Write(LogLevel.Error, tag, message);

    public static void Error(string tag, Exception ex)
        => Write(LogLevel.Error, tag, $"{ex.GetType().Name}: {ex.Message}\n{ex.StackTrace}");

    private static void Write(LogLevel level, string tag, string message)
    {
        if (level < minLevel) return;

        lock (sync)
        {
            if (file == null) Initialize();
            WriteRaw(level, tag, message);
        }
    }

    private static void WriteRaw(LogLevel level, string tag, string message)
    {
        string stamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff");
        string prefix = level switch
        {
            LogLevel.Debug => "DBG",
            LogLevel.Info => "INF",
            LogLevel.Warn => "WRN",
            LogLevel.Error => "ERR",
            _ => "???",
        };

        string line = $"[{stamp}] {prefix} [{tag}] {message}";

        try
        {
            file?.WriteLine(line);
            file?.Flush();
        }
        catch { }

        System.Diagnostics.Debug.WriteLine(line);
    }

    private static void RotateIfNeeded(string path, long maxBytes, long maxFiles)
    {
        try
        {
            if (!File.Exists(path)) return;
            var fi = new FileInfo(path);
            if (fi.Length < maxBytes) return;

            string oldest = $"{path}.{maxFiles}";
            if (File.Exists(oldest)) File.Delete(oldest);

            for (int i = (int)maxFiles - 1; i >= 1; i--)
            {
                string src = $"{path}.{i}";
                string dst = $"{path}.{i + 1}";
                if (File.Exists(src))
                {
                    if (File.Exists(dst)) File.Delete(dst);
                    File.Move(src, dst);
                }
            }

            File.Move(path, $"{path}.1");
        }
        catch { }
    }
}
